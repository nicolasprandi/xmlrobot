-- drop views
DROP VIEW IF EXISTS int.kopf;

-- box
DROP TABLE IF EXISTS int.box;
CREATE TABLE int.box (
                box_id INTEGER NOT NULL AUTO_INCREMENT,
                zeit_import TIMESTAMP,
                zeit_qk TIMESTAMP,
                zeit_export TIMESTAMP,
                qualitaet smallint,
                uri VARCHAR(500),
                beschreibung VARCHAR(500),
                PRIMARY KEY (box_id)
);

-- attr
DROP TABLE IF EXISTS int.attr CASCADE;
CREATE TABLE int.attr (
                attr_id INTEGER NOT NULL AUTO_INCREMENT,
                obj_id INTEGER NOT NULL,
                attr_typ smallint NOT NULL,
                box_id INTEGER NOT NULL,               
                PRIMARY KEY (attr_id)
);


-- wert
DROP TABLE IF EXISTS int.wert CASCADE;
CREATE TABLE int.wert (
                wert_id INTEGER NOT NULL AUTO_INCREMENT,

                bis TIMESTAMP,
                wann TIMESTAMP DEFAULT now() NOT NULL,
                ist_top smallint DEFAULT 0 NOT NULL,
                hat_mehr smallint DEFAULT 0 NOT NULL,
                box_id INTEGER NOT NULL,                   
                PRIMARY KEY (wert_id)
);

CREATE VIEW int.kopf AS
 SELECT n1.obj_id,
    n1.obj_typ,
    n2.kopf
   FROM ( SELECT obj.obj_id,
            obj.obj_typ,
            ot.kopf_attr_typ
           FROM int.obj obj,
            vok.tobj ot
          WHERE obj.obj_typ = ot.obj_typ) n1
     LEFT JOIN ( SELECT attr.obj_id,
            attr.attr_typ,
            wert.inhalt AS kopf
           FROM int.attr attr,
            int.wert wert
          WHERE attr.attr_id = wert.attr_id AND wert.ist_top = 1) n2 ON n1.obj_id = n2.obj_id AND n1.kopf_attr_typ = n2.attr_typ;
