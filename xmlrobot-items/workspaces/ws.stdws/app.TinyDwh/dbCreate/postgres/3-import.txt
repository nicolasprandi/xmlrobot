CREATE VIEW #kopf AS
 SELECT n1.obj_id,
    n1.obj_typ,
    n2.kopf
   FROM ( SELECT obj.obj_id,
            obj.obj_typ,
            ot.kopf_attr_typ
           FROM #obj obj,
            t_o ot
          WHERE obj.obj_typ = ot.obj_typ) n1
     LEFT JOIN ( SELECT attr.obj_id,
            attr.attr_typ,
            wert.inhalt AS kopf
           FROM #attr attr,
            #wert wert
          WHERE attr.attr_id = wert.attr_id AND wert.ist_top = 1) n2 ON n1.obj_id = n2.obj_id AND n1.kopf_attr_typ = n2.attr_typ;
